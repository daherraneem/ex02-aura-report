\chapter{חלופות לתוכנית ההשקה}
\label{ch:alternatives}

חקר השוק בפרק~\ref{ch:market} הציב שלושה נתוני פתיחה להחלטת ההשקה של
\en{AURA}: תמחור יעד של \ltr{69}~ש"ח ל-\ltr{10} מ"ל (ומארז חמשת הגוונים
ב-\ltr{299}~ש"ח) בין טינט הדיסקאונט (כ-\ltr{15}~ש"ח) למותגי הפרימיום המיובאים
(\ltr{99--175}~ש"ח), לצד תחרות מותגי ביניים בפארם (\ltr{40--115}~ש"ח); קהל
ליבה של כ-\ltr{211.6} אלף סטודנטיות ב-\ltr{57} מוסדות (\en{SAM} של \ltr{25.55}
מיליון~ש"ח); ו\textbf{חסם התנסות חושי} במותגי הפרימיום הנרכשים מרחוק. פרק זה
מגבש שלוש חלופות השקה שונות מהותית ל-\ltr{6} חודשים, משווה ביניהן במטריצת
החלטה משוקללת הכוללת סולם עוגנים וניתוח רגישות, מפרט את ארכיטקטורת ההשקה
ואת כלכלת היחידה ודוח הרווח וההפסד (\en{P\&L}) לחודשי המכירה \ltr{4--6},
ומציג את החדשנות הפיג'יטלית (\en{Phygital}) של המותג.

\section{החלופות האסטרטגיות להשקה}
\label{sec:alt-options}

לפי דוח הענף \en{State of Beauty 2025} של \en{McKinsey}, המסחר המקוון מרכז
כ-\ltr{28\%} ממכירות הקוסמטיקה בעולם והמסחר החברתי (\en{Social Commerce}) צומח
בקצב המהיר ביותר, אך מודל \en{D2C} טהור מתקשה לצמוח לבדו בשל עליית עלויות
הפרסום הממומן, ונקודות מגע חווייתיות נותרות חשובות לגילוי מותגים
חדשים~\cite{mckinsey2025beauty}. לפיכך נבחנו שלוש חלופות הנבדלות בתמהיל הערוצים,
במבנה העלויות ובמענה לחסם החושי, תוך שימוש באותו בסיס עלות לאירוע שטח
(כ-\ltr{13.3} אלף~ש"ח ליום פעילות הכולל דוכן, הסעדה מאושרת, צוות ו-\ltr{500}
דוגמיות מיני אטומות).

\subsection{חלופה א': דיגיטל תחילה והשקות מוגבלות}
\label{subsec:alt-digital}

חלופה א' (\en{Digital-First \& Viral Drops}) נשענת על מכירה \textbf{בלעדית באתר
ה-\en{D2C} של \en{AURA}} במהדורות מוגבלות (\en{Drops}) וברשימת המתנה, ללא נוכחות
פיזית או מרקטפלייס. כל תקציב השיווק מופנה ל-\en{TikTok}, למשפיעניות קטנות
(\en{Micro-Influencers}, \ltr{5--30} אלף עוקבות), לסרטוני \en{ASMR}, לתוכן
גולשות (\en{UGC}) ולמדיה ממומנת. תקציב \ltr{6} החודשים הוא הנמוך ביותר ---
כ-\ltr{115} אלף~ש"ח (תשתית אתר, רגולציה ומיתוג \ltr{25} אלף; תוכן ומשפיעניות
\ltr{50} אלף; ומדיה ממומנת וסקרים \ltr{40} אלף~ש"ח) --- ללא עמלות קמעונאיות.

\begin{itemize}
  \item \textbf{יתרונות:} שליטה מלאה במותג ובנתוני צד ראשון; זמן הגעה מהיר לשוק
    (\en{Time-to-Market}) ללא משא ומתן עם מרקטפלייס או אישורי קמפוס; ופוטנציאל
    לתחושת בהילות (\en{FOMO}) ולמרווח תרומה גבוה ליחידה.
  \item \textbf{חסרונות:} אינה מספקת התנסות חושית פיזית --- כפי שמסכמת סקירת
    השיווק החושי של \en{Krishna}, גירויים חושיים (ריח, מגע ומראה) משפיעים על
    תפיסת מוצר והתנהגות צרכנית, ותיווכם דרך מסך בלבד מוגבל~\cite{krishna2012sensory}.
    בנוסף, תלות באלגוריתם ובממומן חושפת מותג חדש לעלות רכישת לקוחה (\en{CAC})
    גבוהה ולהמרה נמוכה יותר.
\end{itemize}

\subsection{חלופה ב': קמפוס וחוויה תחילה}
\label{subsec:alt-campus}

חלופה ב' (\en{Campus \& Experience-First}) מקצה כ-\ltr{70\%} מתקציב השיווק
לשמונה ימי הפעלה ניידים בארבעה קמפוסים מרכזיים (חיפה, תל אביב, בן-גוריון
והעברית, שני ימים בכל קמפוס באותו בסיס עלות של כ-\ltr{13.3} אלף~ש"ח ליום,
ובנוסף שת"פי בתי קפה), בעוד הרשתות משמשות לתיעוד השטח. התקציב הכולל הוא הגבוה
ביותר --- כ-\ltr{195} אלף~ש"ח (\ltr{107} אלף לימי קמפוס ודוגמיות, \ltr{33} אלף
לשת"פי בתי קפה ולוגיסטיקה, ו-\ltr{55} אלף לאתר בסיסי, רגולציה ותיעוד חברתי).

\begin{itemize}
  \item \textbf{יתרונות:} מענה ישיר לחסם החושי בארבעה מוסדות גדולים, המרה גבוהה
    במפגש פנים אל פנים, ובניית קהילה מקומית סביב פרסונת ``נור''.
  \item \textbf{חסרונות:} מדרגיות (\en{Scalability}) איטית ומוגבלת גיאוגרפית;
    עלות קבועה גבוהה הפוגעת בהחזר ההשקעה (\en{ROI}); וסיכון עיכובים באישורים
    מול ארבע הנהלות קמפוסים.
\end{itemize}

\subsection{חלופה ג': אומניצ'אנל היברידי ממוקד}
\label{subsec:alt-hybrid}

חלופה ג' (\en{Omnichannel Hybrid}) מיישמת את המלצת דוח \en{McKinsey} לאקוסיסטם
משולב~\cite{mckinsey2025beauty}: השקה דו-ערוצית מחודש \ltr{4} באתר ה-\en{D2C}
של \en{AURA} \textbf{במקביל} לפלטפורמת מסחר מקוונת מובילה בישראל ---
\en{Terminal X} (קטגוריית ביוטי עם משלוח מהיום למחר~\cite{terminalx2026beauty})
או \en{Super-Pharm Online}~\cite{superpharm2026marketplace} (הפלטפורמה תינעל
בשלב \en{WP1}). את המכירה מגבים \ltr{20--30} משפיעניות קטנות (\ltr{5--30} אלף
עוקבות, הנהנות משיעורי מעורבות גבוהים יותר מחשבונות גדולים לפי דוח הענף של
\en{HypeAuditor}~\cite{hypeauditor2025state}), \ltr{3} ברי טעימות בקמפוסים
(\ltr{500} דוגמיות מיני אטומות בכל אירוע), ושאלון התאמת טעם וגוון באתר ה-\en{D2C}.
התקציב מאוזן ועומד על \ltr{150} אלף~ש"ח: \ltr{78} אלף (\ltr{52\%}) לתוכן,
משפיעניות (\ltr{43K}), מדיה ממומנת וסקרים (\ltr{35K}); \ltr{40} אלף
(\ltr{27\%}) לשלושת ברי הטעימות (כ-\ltr{13.3K} לאירוע); ו-\ltr{32} אלף
(\ltr{21\%}) לתשתית \en{D2C}, שאלון, מרקטפלייס ורגולציה (\ltr{10K}).

\begin{itemize}
  \item \textbf{יתרונות:} אמון צרכני ומשלוח מהיר במרקטפלייס, פתרון מדוד לחסם
    החושי בשלושה קמפוסים, ושימור מרווח מלא במארזי ה-\en{D2C}.
  \item \textbf{חסרונות:} עמלת מרקטפלייס (הנחת תכנון \ltr{20\%} על \ltr{35\%}
    מהנפח), תלות באישור הצטרפות לפלטפורמה, ומורכבות תפעולית גבוהה יותר.
\end{itemize}

\section{מטריצת החלטה משוקללת וניתוח רגישות}
\label{sec:alt-matrix}

להשוואה מובנית יושמה \textbf{מטריצת ניקוד משוקללת} (\en{Concept-Scoring Matrix})
לפי \en{Ulrich}, \en{Eppinger} ו-\en{Yang}~\cite{ulrich2020product}. הציונים הם
\textbf{הערכת מומחה סובייקטיבית של הצוות} בסולם \ltr{1--10} המעוגן בשלוש דרגות:
\ltr{3} = עמידה חלשה בדרישות, \ltr{6} = עמידה חלקית, \ltr{9} = עמידה גבוהה.

\begin{definitionbox}[נוסחת הציון המשוקלל]
עבור \ltr{6} קריטריונים בעלי משקלים $w_i$ ($\sum_{i=1}^{6} w_i = 1.00$), הציון
המשוקלל $S_j$ של חלופה $j \in \{1, 2, 3\}$ לפי ציוניה $r_{i,j} \in [1, 10]$
הוא~\cite{ulrich2020product}:
\begin{equation}
  S_j = \sum_{i=1}^{6} w_i \cdot r_{i,j}, \qquad \sum_{i=1}^{6} w_i = 1.00,
  \qquad j^{*} = \arg\max_{j} S_j
  \label{eq:weighted-score}
\end{equation}
\end{definitionbox}

המשקלים נגזרו מיעדי פרק~\ref{ch:problem} ומממצאי פרק~\ref{ch:market}: התאמה
לפרסונה ולתובנה החושית (\ltr{0.25}), ויראליות ומודעות (\ltr{0.20}), יעילות
תקציבית והחזר (\ltr{0.20}), זמן הגעה לשוק (\ltr{0.15}), המרה למכירות ראשוניות
(\ltr{0.10}), ושליטה במותג ובנתוני צד ראשון (\ltr{0.10}). טבלה~\ref{tab:decision-matrix}
מציגה ניקוד שמרני המונע הטיה לטובת חלופה ג': חלופה ג' מדורגת \ltr{8} בהתאמה
חושית (מתחת ל-\ltr{9} של חלופה ב' הפרוסה על \ltr{8} ימי קמפוס), \ltr{8}
בוויראליות (שווה לחלופה א') ו-\ltr{8} בהמרה (שווה לחלופה ב').

\section{ארכיטקטורת ההשקה, כלכלת היחידה ורווח והפסד}

כדי לעגן את יעד המכירות לחודשים \ltr{4--6} בנתונים כלכליים שקופים, הוגדרו
\textbf{כלכלת היחידה} ו\textbf{משפך ההמרה לערוצים} תוך ניכוי \ltr{18\%} מע"מ
החל בישראל מינואר \ltr{2025}~\cite{pwc2025israelvat}. במחיר צרכני של
\ltr{69.0}~ש"ח ליחידה פיזית, ההכנסה נטו היא \ltr{58.5}~ש"ח ($69/1.18$). תחת
הנחות תכנון מפורשות --- עלות ייצור ואריזה (\en{COGS}) של \ltr{14.0}~ש"ח, שילוח
וסליקה נטו של \ltr{6.5}~ש"ח ועמלת מרקטפלייס משוקללת של \ltr{4.0}~ש"ח (\ltr{20\%}
עמלה על \ltr{35\%} מהנפח; \ltr{0\%} ב-\en{D2C} ובבר) --- העלות המשתנה היא
\ltr{24.5}~ש"ח ו\textbf{מרווח התרומה המשוקלל הוא \ltr{34.0}~ש"ח ליחידה}
(\ltr{58.1\%} מההכנסה נטו). לכיסוי תקציב ההשקה של \ltr{150} אלף~ש"ח (ללא שכר

\section{החדשנות בגישת ההשקה של המותג}
\label{sec:alt-innovation}

החדשנות של \en{AURA} מחברת בין קולינריה לביוטי באמצעות \textbf{חוויית השקה
פיג'יטלית} (\en{Phygital}). ספרות הקמעונאות של \en{Alexander}
ו-\en{Varley}~\cite{alexander2025phygital} ושל \en{Banik}~\cite{banik2021phygital}
מראה כי שילוב מרחב חווייתי זמני עם ממשק דיגיטלי אינטראקטיבי מגביר מעורבות
צרכנית ונכונות לקנייה. בהתאם לכך משלבת \en{AURA} שני רכיבים משלימים: \textbf{בר
טעימות בקמפוס} ו-\textbf{שאלון התאמת טעם וגוון}.

\subsection{שלב פיזי: בר הטעימות בקמפוס}
\label{subsec:innov-bar}

שלושת ברי הטעימות (\en{The AURA Tasting Bar}) מופעלים כעמדת טעימות חמה בשיתוף
ספק הסעדה מורשה ועגלת קפה בקמפוס, ומתוכננים לתפוקה ריאלית של \ltr{500} דוגמיות
מיני אטומות ליום אירוע (שתי עמדות שירות במשך \ltr{6} שעות, כ-\ltr{42} מבקרות
לשעה לעמדה). ההשערה השיווקית שתיבחן בהשקה היא כי טעימה קולינרית זעירה מחמשת
עולמות הטעם (\en{Golden Honey Glaze}, \en{Velvet Peach Nectar}, \en{Berry Crush
Elixir}, \en{Ruby Pomegranate Jewel} ו-\en{Silk Cocoa Glaze}) יחד עם קבלת
דוגמית מיני אישית אטומה ואפליקטור חד-פעמי משפרות את נכונות הניסיון והשיתוף
ב-\en{TikTok}~\cite{voguehk2025food,glossy2025lips}, במחיר ובזמינות מקומית
ש-\en{Rhode} אינה מציעה בישראל~\cite{rhode2026tint}. על כרטיס הטעימות מוטבע
קוד \en{QR} המשויך לקוד המעקב של \en{G5} באתר ה-\en{D2C}.

\subsection{שלב דיגיטלי: שאלון התאמת טעם וגוון}
\label{subsec:innov-ai}

לגולשות מחוץ לשלושת הקמפוסים יוטמע באתר ה-\en{D2C} \textbf{שאלון התאמה
קולינרי-גוון} (\en{Flavor \& Shade Quiz}) בן שלוש שאלות: שתי שאלות טעם
ואווירה לצורך סיפור המותג, ושאלת התאמת גוון עור המלווה בצילומי הדגמה
(\en{Swatches}) על \ltr{3--4} גווני עור שונים כדי למנוע טעויות בחירה והחזרות.
בניגוד לבר הטעימות הפיזי, השאלון אינו מחליף התנסות חושית מלאה אלא מתפקד
כ\textbf{כלי המרה ואיסוף נתוני צד אפס} (השערה שתיבחן בבדיקת \en{A/B} באתר).
בסיומו מפיק השאלון המלצת גוון וקישור ישיר לסל, לצד כרטיס ``קבלת בית קפה''
אישי לשיתוף בסטורי. לפי סקר של חברת הייעוץ \en{McKinsey}, \ltr{71\%} מהצרכנים
מצפים לאינטראקציה מותאמת אישית והתאמה אפקטיבית מניבה לעיתים תוספת הכנסות של
\ltr{10\%--15\%}~\cite{mckinsey2021personalization}; ובדומה לכך, מקרה בוחן
שפרסמה ספקית התוכנה \en{Octane AI} מדגים שימוש בשאלוני ביוטי לאיסוף נתוני צד
אפס (תוך מודעות למגבלה כי מדובר בנתוני ספק ולא במחקר מבוקר)~\cite{octaneai2025quiz}.
