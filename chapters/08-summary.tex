\chapter{סיכום}
\label{ch:summary}

\textbf{הממצאים העיקריים.} הדוח בחן את השקת \en{AURA} בישראל לאורך \ltr{6}
חודשים. המגמות --- ביוטי בהשראת מזון ומסחר חברתי --- תומכות ברעיון
(פרק~\ref{ch:intro}), אך מותג חדש מתמודד עם שלושה פערים: נראות, התנסות ומחיר
(פרק~\ref{ch:problem}). הספרות מראה שכל אחד מכלי ההשקה --- גירויים חושיים, שמות
מזון, משפיעניות ודוגמיות --- אפקטיבי בנפרד, אך שילובם לא נבדק
(פרק~\ref{ch:literature}). חקר השוק זיהה חלל \en{Masstige} סביב \ltr{69} ש"ח ויעד
ריאלי של \ltr{255} אלף ש"ח בחלון ההשקה (פרק~\ref{ch:market}). החלופה ההיברידית
--- אתר \en{D2C}, מרקטפלייס ישראלי ו-\ltr{3} ברי טעימות בקמפוסים --- קיבלה את
הציון הגבוה במטריצת ההחלטה (פרק~\ref{ch:alternatives}), תורגמה לשש חבילות עבודה
בתקציב של \ltr{150} אלף ש"ח (פרק~\ref{ch:workplan}), ותוכנית ההפחתה מורידה את
החשיפה המרבית לסיכון מ-\ltr{16} ל-\ltr{6} (פרק~\ref{ch:risks}).

\textbf{מדוע התוכנית מוכנה לביצוע.} לכל יעד יש מדד, מועד ואחראי (\en{G1--G5});
לכל דרישה יש ערך יעד ושיטת אימות (\en{REQ-01--REQ-11}); ולכל רכיב של השפעת
הדוגמיות יש דרך מדידה, לפי מודל מאמר הדגל. הסיכון הקריטי --- הודעות השיווק
למשרד הבריאות --- נמצא בנתיב הקריטי, ושער \en{Go/No-Go} בסוף חודש \ltr{3}
(\en{M3}) מונע עלייה לאוויר לפני שהוא נסגר.

\textbf{הצעדים המיידיים (חודש \ltr{1}).}
\begin{enumerate}
  \item הגשת הודעות השיווק לחמשת הגוונים למשרד הבריאות (\en{REQ-01}, אבן דרך \en{M1}).
  \item אישור ספר המותג והאריזות, ופתיחת תהליך ההצטרפות למרקטפלייס.
  \item פנייה ל-\ltr{3} הקמפוסים ולמאגר היוצרות, וניסוח כללי סימון הפרסומת (\en{REQ-02}).
  \item הכנת סקר המודעות הבסיסי (\en{REQ-09}), שמולו יימדד יעד \en{G1}.
\end{enumerate}

\begin{keymessage}[המסר המרכזי למנהלים]
\en{AURA} יכולה להפוך ממותג שאיש אינו מכיר למותג שנמכר בתוך \ltr{6} חודשים, אם
תחבר תוכן ברשתות להתנסות אמיתית בקמפוס. התוכנית מוכנה: היעדים מדידים, התקציב
מוגדר, והסיכון הקריטי מבוקר --- הצעד הראשון הוא הגשת ההודעות למשרד הבריאות.
\end{keymessage}
