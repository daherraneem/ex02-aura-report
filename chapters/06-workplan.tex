\chapter{תוכנית עבודה}
\label{ch:workplan}

פרק זה מתרגם את חלופת האומניצ'אנל ההיברידית מפרק~\ref{ch:alternatives} (תקציב
השקה של \ltr{150}~אלף~ש"ח, יעד של \ltr{2,200} יחידות פיזיות בחודשי המכירה
\ltr{4--6}) ואת יעדי התכן ומפרט הדרישות מפרק~\ref{ch:problem} (\en{G1--G5}
ו-\en{REQ-01--REQ-11}) לתוכנית ביצוע בת \ltr{6} חודשים המעוגנת בלוח השנה
האקדמי בישראל: דצמבר \ltr{2026} עד מאי \ltr{2027} (כך שחודש \ltr{4}, מרץ
\ltr{2027}, נפתח עם סמסטר ב' בתנועת סטודנטיות גבוהה ולא בתקופת מבחני פברואר).
בהתאם למדריך \en{PMBOK} ול-\en{Kerzner}~\cite{pmi2021pmbok,kerzner2022project},
התוכנית נשענת על מבנה פירוק עבודה (\en{WBS}) ותקציב, מטריצת אחריות (\en{RACI}),
ותרשים גאנט עם אבני דרך ונתיב קריטי.

\section{מבנה פירוק העבודה ותקציב ההשקה}
\label{sec:workplan-wbs}

מבנה פירוק העבודה (\en{WBS}) מפרק את היקף ההשקה לשש חבילות מונחות תוצרים
(\en{WP1} עד \en{WP6}) תוך שמירה על סדר הנדסי-רגולטורי תקין: \textbf{נעילת
פורמולה וקבוצות מיקוד בחודש \ltr{1} $\leftarrow$ הגשת הודעות משרד הבריאות
(\en{REQ-01}) וייצור המלאי בחודש \ltr{2} $\leftarrow$ קבלת האישורים ובדיקות
המערכת בחודש \ltr{3}} (כדי למנוע שינוי פורמולה לאחר ההודעה שיאלץ הגשה חוזרת).
סבב הייצור הראשוני כולל \ltr{3,000} יחידות מכירה (\ltr{600} לכל גוון, כולל
מלאי ביטחון מעל יעד \ltr{2,200} היחידות) ו-\ltr{1,500} דוגמיות מיני אטומות
(\ltr{500} לכל בר טעימות), וכולל בדיקת סימן מסחר לשם \en{AURA} כבר ב-\en{WP1}.
איור~\ref{fig:wbs-tree} מציג את עץ ה-\en{WBS} המלא.

\section{לוח זמנים, אבני דרך ונתיב קריטי}
\label{sec:workplan-schedule}

לוח הזמנים נבנה לפי \textbf{שיטת הנתיב הקריטי} (\en{Critical Path Method}) של
\en{Kelley} ו-\en{Walker}~\cite{kelley1959critical}, המזהה את שרשרת המשימות
התלויות שמרווח הגמישות (\en{Float}) שלה הוא אפס. איור~\ref{fig:gantt-launch}
מציג את תרשים הגאנט (דצמבר \ltr{2026} -- מאי \ltr{2027}; ציר הזמן נקרא משמאל
לימין), כולל קו ה-\en{Go-Live} בתחילת מרץ \ltr{2027} (פתיחת סמסטר ב') ושש
אבני הדרך (\en{M1--M6}).

\textbf{ניתוח הנתיב הקריטי.} מתרשים הגאנט (איור~\ref{fig:gantt-launch}) עולה כי
הנתיב הקריטי עובר ברצף: \textbf{משימה \ltr{1.1}} (נעילת פורמולה וקבוצות מיקוד,
דצמבר) $\leftarrow$ \textbf{משימה \ltr{1.3}} (הגשה ואישור \en{REQ-01}, ינואר--פברואר)
$\leftarrow$ \textbf{שער \en{M3}} $\leftarrow$ \textbf{משימה \ltr{4.1}} (השקה
ושער מסחרי \en{M4}, מרץ) $\leftarrow$ \textbf{משימה \ltr{5.1}} (אופטימיזציה,
אפריל) $\leftarrow$ \textbf{משימה \ltr{6.1}} (שימור ודוח \en{P\&L}, מאי). בשל
חובת הודעת השיווק לפני מכירה~\cite{ita2023israelcosmetics}, לכל עיכוב בנעילת
הפורמולה או ב-\en{REQ-01} יש אפס מרווח גמישות והוא ידחה את פתיחת סמסטר ב';
לעומת זאת, להפקת התוכן, לגיוס היוצרות ולאישורי הקמפוס יש מרווח של \ltr{2--3}
שבועות.

\begin{keymessage}[המסר המרכזי מתוכנית העבודה]
תוכנית העבודה מפרקת את אסטרטגיית ההשקה לשש חבילות עבודה (\en{WP1--WP6})
המעוגנות בלוח השנה האקדמי (דצמבר \ltr{2026} -- מאי \ltr{2027}) במסגרת תקציב
של \ltr{150} אלף ש"ח (כולל סעיף רגולציה ייעודי של \ltr{10} אלף ש"ח). הקדמת
נעילת הפורמולה לפני הגשת \en{REQ-01} וקביעת שער מסחרי ב-\en{M4} מגינים על
הנתיב הקריטי ועל מסגרת ה-\en{P\&L} --- ממצאים המנחים את ניתוח הסיכונים
בפרק~\ref{ch:risks}.
\end{keymessage}
