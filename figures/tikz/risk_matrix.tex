\begin{tikzpicture}[

    score/.style={font=\sffamily\fontsize{6.2}{7}\selectfont, text=AuraMuted!85},

      \fi\fi\fi

  }

  % Dashed boundary for R >= 12 mandatory mitigation zone
  \draw[AuraRuby!85!AuraPlum, line width=1.2pt, dashed]
    ({2*\cw}, {5*\ch}) -- ({2*\cw}, {3*\ch}) -- ({3*\cw}, {3*\ch}) --

  \node[axlbl] at ({2.5*\cw}, -0.62) {\texthebrew{השפעה (\textenglish{I}: \LR{1} זניחה -- \LR{5} קריטית)}};
  \node[axlbl, rotate=90] at (-0.72, {2.5*\ch}) {\texthebrew{הסתברות (\textenglish{P}: \LR{1--5})}};

  % Plot R1-R8 cleanly without overlap:
  % Cell (4,4): R1, R2

  \node[critbadge] at ({3.72*\cw}, {3.42*\ch}) {R2};
  % Cell (5,3): R3
  \node[critbadge] at ({4.48*\cw}, {2.44*\ch}) {R3};
  % Cell (3,4): R4

  % Cell (4,2): R8
  \node[rbadge] at ({3.48*\cw}, {1.44*\ch}) {R8};

  % Legend on the right
  \begin{scope}[shift={({5*\cw + 0.50}, 0.22)}]
